def rpw():
    a = string.ascii_letters + string.digits
    return "T1633-" + "".join(secrets.choice(a) for _ in range(20)) + "-aA1!"

PW = rpw()

def ps(s, show=True):
    rc, out = dc.ps(s)
    if show and out.strip():
        print(out.strip())
    return rc, out

def admin_conn():
    tls = Tls(validate=ssl.CERT_REQUIRED, ca_certs_file=REPO + "/lab/windows/.win-lab-ca.pem", valid_names=["dc1.win.activediq.test", addr])
    srv = Server(addr, port=636, use_ssl=True, tls=tls, get_info=ldap3.NONE)
    return srv

def bind(user):
    srv = admin_conn()
    c = Connection(srv, user=user + "@" + DOM, password=PW, auto_bind=True)
    return c

def adminbind():
    srv = admin_conn()
    return Connection(srv, user="Administrator@" + DOM, password=env["WIN_LAB_ADMIN_PASSWORD"], auto_bind=True)

R2_ATTRS = ["objectSid", "sIDHistory", "userAccountControl", "primaryGroupID", "memberOf", "lastKnownParent",
            "msDS-AllowedToDelegateTo", "msDS-AllowedToActOnBehalfOfOtherIdentity", "msDS-SecondaryKrbTgtNumber",
            "msDS-LastKnownRDN", "adminCount", "isCriticalSystemObject", "nTSecurityDescriptor",
            "isDeleted", "isRecycled", "name", "description", "objectGUID"]

def guidhex(g):
    return uuid.UUID(g).bytes_le.hex()

def read(conn, g, label, attrs=R2_ATTRS, scope=BASE, base=None, filt="(objectClass=*)", sd=True):
    ctl = [SHOWDEL]
    if sd:
        ctl.extend(security_descriptor_control(criticality=False, sdflags=0x07))
    b = base or "<GUID=%s>" % guidhex(g)
    try:
        ok = conn.search(b, filt, search_scope=scope, attributes=attrs, controls=ctl)
    except Exception as e:  # noqa
        print("  %-52s EXC %s" % (label, str(e)[:120])); return None
    r = conn.result
    ents = [e for e in conn.response if e.get("type") == "searchResEntry"]
    got = {}
    if ents:
        for k, v in ents[0]["raw_attributes"].items():
            if v:
                got[k] = len(v)
    print("  %-52s rc=%s %s entries=%d attrs_returned=%s" % (label, r["result"], r["description"], len(ents), sorted(got) if ents else "-"))
    return ents

def mkvictims():
    # v1 baseline defaults; v2 OU-inherited GenericRead for cls2/cls3 (added before); v3 explicit Full Control ACE for cls2 on the object
    sc = r'''
Import-Module ActiveDirectory
$base='%(base)s'; $ou='%(ou)s'
$pw = ConvertTo-SecureString '%(pw)s' -AsPlainText -Force
$out=@()
foreach($n in 'v1','v2','v3'){
  $u = New-ADUser -Name "t1633-$n" -SamAccountName "t1633-$n" -Path $ou -AccountPassword $pw -Enabled $true -PassThru
  Add-ADGroupMember -Identity G1633 -Members $u
  Set-ADUser -Identity $u -Replace @{adminCount=1; description='victim'}
  Set-ADUser -Identity $u -Add @{'msDS-AllowedToDelegateTo'='cifs/x.%(dom)s'}
  $out += "$n " + (Get-ADUser $u -Properties objectGUID).ObjectGUID
}
$out
''' % dict(base=BASE_DN, ou=OU, pw=PW, dom=DOM)
    rc, out = ps(sc, show=False)
    d = {}
    for l in out.splitlines():
        p = l.split()
        if len(p) == 2 and p[0] in ("v1", "v2", "v3"):
            d[p[0]] = p[1]
    return d
