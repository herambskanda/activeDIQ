def guidhex(g):

def sdc(flags):
    return security_descriptor_control(criticality=False, sdflags=flags)


def sddl(b):
    """Decode a binary SD to SDDL on dc1 (RawSecurityDescriptor)."""
    rc, o = ps("$b=[Convert]::FromBase64String('%s');$s=New-Object Security.AccessControl.RawSecurityDescriptor($b,0);$s.GetSddlForm('All')" % base64.b64encode(b).decode())
    return o.strip()


def read_sd(conn, base, flags=0x7, showdel=True):
    ctl = ([SHOWDEL] if showdel else []) + sdc(flags)
    conn.search(base, "(objectClass=*)", search_scope=BASE, attributes=["nTSecurityDescriptor"], controls=ctl)
    ents = [e for e in conn.response if e.get("type") == "searchResEntry"]
    if not ents:
        return conn.result["description"], None
    v = ents[0]["raw_attributes"].get("nTSecurityDescriptor") or []
    return conn.result["description"], (v[0] if v else None)


def read(conn, base, label, attrs, sd=0x7, showdel=True, link=True, quiet=False):
    ctl = []
    if showdel:
        ctl.append(SHOWDEL)
    if link:
        ctl.append(D2065)
    if sd:
        ctl += sdc(sd)
    try:
        conn.search(base, "(objectClass=*)", search_scope=BASE, attributes=attrs, controls=ctl)
    except Exception as e:  # noqa
        out("  %-58s EXC %s" % (label, str(e)[:140])); return None
    ents = [e for e in conn.response if e.get("type") == "searchResEntry"]
    got = {}
    if ents:
        for k, v in ents[0]["raw_attributes"].items():
            if v:
                got[k] = v
    if not quiet:
        out("  %-58s rc=%s %s entries=%d attrs=%s" % (label, conn.result["result"], conn.result["description"], len(ents), sorted(got) if ents else "-"))
    return got if ents else None


def baseline_container():
    a = adminbind()
    d, b = read_sd(a, CONT, 0x7)
    a.unbind()
    return b


def set_sd(conn, dn, data, flags):
    ok = conn.modify(dn, {"nTSecurityDescriptor": [(MODIFY_REPLACE, [data])]}, controls=[SHOWDEL] + security_descriptor_control(criticality=True, sdflags=flags))
    return ok, conn.result["description"]


def transform_sd(cur, rules, owner_sid, strip_trustee_like=None):
    """Typed edit on dc1: load the binary SD, optionally drop ACEs of listed SIDs, add rules [(sid, 'ListChildren,ReadControl')]
    with inheritance None, set the owner. Returns (sddl, new binary)."""
    adds = "\n".join("$r=New-Object System.DirectoryServices.ActiveDirectoryAccessRule([Security.Principal.SecurityIdentifier]'%s',[System.DirectoryServices.ActiveDirectoryRights]'%s','Allow',[System.DirectoryServices.ActiveDirectorySecurityInheritance]'None');$sd.AddAccessRule($r)" % (s, r) for s, r in rules)
    strip = ""
    if strip_trustee_like:
        strip = "$sd.GetAccessRules($true,$false,[Security.Principal.SecurityIdentifier]) | Where-Object { @(%s) -contains $_.IdentityReference.Value } | ForEach-Object { [void]$sd.RemoveAccessRuleSpecific($_) }" % ",".join("'%s'" % s for s in strip_trustee_like)
    rc, o = ps(r'''
$b=[Convert]::FromBase64String('%s')
$sd=New-Object System.DirectoryServices.ActiveDirectorySecurity
$sd.SetSecurityDescriptorBinaryForm($b)
%s
%s
$sd.SetOwner([Security.Principal.SecurityIdentifier]'%s')
$sd.GetSecurityDescriptorSddlForm('All')
[Convert]::ToBase64String($sd.GetSecurityDescriptorBinaryForm())
''' % (base64.b64encode(cur).decode(), strip, adds, owner_sid))
    lines = [l for l in o.strip().splitlines() if l.strip()]
    return lines[-2], base64.b64decode(lines[-1])


def sid_of(sam):
    rc, o = ps("Import-Module ActiveDirectory; (Get-ADObject -Filter \"samAccountName -eq '%s'\" -Properties objectSid).objectSid.Value" % sam)
    return o.strip().splitlines()[-1].strip()


def cleanup_names(ou_dns, users, groups):
    sc = "Import-Module ActiveDirectory\n"
    for ou in ou_dns:
        sc += """$ou='%s'
if (Get-ADOrganizationalUnit -Filter "distinguishedName -eq '$ou'" -ErrorAction SilentlyContinue) {
  Get-ADOrganizationalUnit -SearchBase $ou -Filter * | ForEach-Object { Set-ADOrganizationalUnit -Identity $_ -ProtectedFromAccidentalDeletion $false }
  Remove-ADOrganizationalUnit -Identity $ou -Recursive -Confirm:$false
}
""" % ou
    for u in users:
        sc += "Get-ADUser -Filter \"samAccountName -eq '%s'\" | Remove-ADUser -Confirm:$false\n" % u
    for g in groups:
        sc += "Get-ADGroup -Filter \"samAccountName -eq '%s'\" | Remove-ADGroup -Confirm:$false\n" % g
    sc += "'cleanup done'"
    return ps(sc)[1].strip()


def owner_only_sd(sid):
    rc, o = ps("$sd=New-Object System.DirectoryServices.ActiveDirectorySecurity;$sd.SetOwner([Security.Principal.SecurityIdentifier]'%s');[Convert]::ToBase64String($sd.GetSecurityDescriptorBinaryForm())" % sid)
    return base64.b64decode(o.strip().splitlines()[-1].strip())


def takeover(a, da_sid):
    """Typed owner-only write (SD_FLAGS 0x1) of Domain Admins as owner; falls back to dsacls /takeownership. Returns how it went."""
    nb = owner_only_sd(da_sid)
    ok, r = set_sd(a, CONT, nb, 0x1)
    if ok:
        return "typed owner-only write 0x1: ok"
    o = ps('dsacls "%s" /takeownership' % CONT)[1].strip()[-80:]
    return "typed owner-only write 0x1 REFUSED (%s); dsacls /takeownership: %s" % (r, o)


def dsacls_text(extra=""):
    return ps('dsacls "%s"' % CONT)[1].strip()


def guid_of(filt, base=BASE_DN):
    rc, o = ps("Import-Module ActiveDirectory; (Get-ADObject -Filter \"%s\" -SearchBase '%s').ObjectGUID.ToString()" % (filt, base))
    return o.strip().splitlines()[-1].strip()

def sid_str(b):
    rev, n = b[0], b[1]
    ia = int.from_bytes(b[2:8], "big")
    subs = struct.unpack("<%dI" % n, b[8:8 + 4 * n])
    return "S-%d-%d-%s" % (rev, ia, "-".join(str(x) for x in subs)), 8 + 4 * n

def parse_sd(b):
    rev, _, ctl, oo, og, os_, od = struct.unpack("<BBHIIII", b[:20])
    r = {"control": hex(ctl), "owner": sid_str(b[oo:])[0] if oo else None, "aces": []}
    if od:
        _, _, size, cnt, _ = struct.unpack("<BBHHH", b[od:od + 8])
        p = od + 8
        for _ in range(cnt):
            t, f, sz, mask = struct.unpack("<BBHI", b[p:p + 8])
            q = p + 8; obj = inh = None
            if t in (5, 6, 7, 8, 0xB, 0xC, 0xD, 0xE, 0xF, 0x10, 0x11, 0x12, 0x13):  # object-type ACEs
                oflags = struct.unpack("<I", b[q:q + 4])[0]; q += 4
                if oflags & 1: obj = str(uuid.UUID(bytes_le=b[q:q + 16])); q += 16
                if oflags & 2: inh = str(uuid.UUID(bytes_le=b[q:q + 16])); q += 16
            sid = sid_str(b[q:p + sz])[0]
            r["aces"].append((t, f, mask, obj, inh, sid))
            p += sz
    return r
