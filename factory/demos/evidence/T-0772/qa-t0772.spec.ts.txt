const uniq = () => `qa-${randomBytes(4).toString("hex")}`;

async function va(context: BrowserContext, page: Page, uv = true) {
  const cdp = await context.newCDPSession(page);
  await cdp.send("WebAuthn.enable");
  const { authenticatorId } = await cdp.send("WebAuthn.addVirtualAuthenticator", { options: { protocol: "ctap2", transport: "internal", hasResidentKey: true, hasUserVerification: true, isUserVerified: uv, automaticPresenceSimulation: true } });
  return { cdp, authenticatorId };
}
async function hold(cdp: any, authenticatorId: string, key: Passkey) {
  await cdp.send("WebAuthn.addCredential", { authenticatorId, credential: { credentialId: key.credentialId.toString("base64"), isResidentCredential: true, rpId: new URL(base).hostname, privateKey: key.privateKey.export({ type: "pkcs8", format: "der" }).toString("base64"), userHandle: key.userHandle.toString("base64"), signCount: 0 } });
}
async function acct(key?: Passkey) { const a = await createLocalAccount(uniq()); if (key) await addPasskey(a.id, key); return a; }
async function pw(page: Page, a: { username: string; password: string }) {
  await page.goto("/directory/users");
  await page.getByLabel("Account name").fill(a.username);
  await page.getByLabel("Password").fill(a.password);
  await page.getByRole("button", { name: "Continue" }).click();
  await expect(page.getByLabel("Authenticator code")).toBeVisible();
}
const sess = async (c: BrowserContext) => (await c.cookies()).filter((x) => x.name === "__Host-adiq_session").length;

test("K1 keyboard-only sign-in with a key + focus order + screenshots", async ({ page, context }) => {
  const key = newPasskey(); const a = await acct(key);
  const { cdp, authenticatorId } = await va(context, page); await hold(cdp, authenticatorId, key);
  await page.goto("/directory/users");
  await page.screenshot({ path: `${ev}/01-password-step.png` });
  const focus = async () => page.evaluate(() => { const e = document.activeElement as HTMLElement; const cs = getComputedStyle(e); return `${e.tagName}:${e.getAttribute("aria-label") || e.textContent?.trim() || (e as HTMLInputElement).name || e.id}|outline=${cs.outlineStyle} ${cs.outlineWidth}|shadow=${cs.boxShadow.slice(0,40)}`; });
  note("K1 initial focus: " + await focus());
  await page.keyboard.type(a.username); await page.keyboard.press("Tab");
  note("K1 after Tab 1: " + await focus());
  await page.keyboard.type(a.password); await page.keyboard.press("Enter");
  await expect(page.getByLabel("Authenticator code")).toBeVisible();
  await page.waitForTimeout(500);
  note("K1 code step focus after 500ms: " + await focus());
  await page.screenshot({ path: `${ev}/02-code-step-key-button.png` });
  await page.keyboard.press("Tab"); note("K1 code step Tab1: " + await focus());
  await page.keyboard.press("Tab"); note("K1 code step Tab2: " + await focus());
  await page.screenshot({ path: `${ev}/03-focus-key-button.png` });
  const f = await focus(); if (!f.includes("security key")) { await page.keyboard.press("Tab"); note("K1 Tab3: " + await focus()); }
  await page.keyboard.press("Enter");
  await expect(page).toHaveURL(/\/directory\/users/);
  expect(await sess(context)).toBe(1);
  await page.waitForTimeout(500);
  await page.screenshot({ path: `${ev}/05-signed-in-landing.png` });
  note("K1 keyboard-only key sign-in OK; landing text: " + (await page.locator("main").innerText()).slice(0, 200).replace(/\n/g, " | "));
});

test("K7 server errors on challenge/verify (route fault injection)", async ({ page, context }) => {
